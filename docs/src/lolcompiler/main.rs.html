1//! lolcompiler: compiles a LOLCODE Markdown (.lol) file into HTML.
2//!
3//! Usage: `lolcompiler input.lol`
4//!
5//! Steps:
6//! 1. Check there is one argument and that it ends in `.lol`.
7//! 2. Read the file and run the compiler (lexer -> parser -> scope
8//!    check -> HTML). Any error is printed and the program exits
9//!    before any file is written.
10//! 3. Write `input.html` next to the input file and open it in the
11//!    default browser.
12
13mod ast;
14mod lexer;
15mod parser;
16mod semantic;
17mod traits;

19use parser::LolCompiler;
20use std::env;
21use std::fs;
22use std::path::Path;
23use std::process::{self, Command};
24use traits::Compiler;
25
26/// Print how to use the program and exit.
27fn usage_error(msg: &str) -> ! {
28    eprintln!("Usage error: {}", msg);
29    eprintln!("Usage: lolcompiler <input.lol>");
30    process::exit(1);
31}
32
33/// Open the HTML file in the default browser for this OS.
34fn open_in_browser(path: &Path) {
35    let p = path.to_string_lossy().to_string();
36    let result = if cfg!(target_os = "macos") {
37        Command::new("open").arg(&p).spawn()
38    } else if cfg!(target_os = "windows") {
39        // "start" is part of cmd, the empty "" is the window title
40        Command::new("cmd").args(["/C", "start", "", &p]).spawn()
41    } else {
42        Command::new("xdg-open").arg(&p).spawn()
43    };
44    if result.is_err() {
45        eprintln!("Note: could not open a browser. Open {} by hand.", p);
46    }
47}
48
49fn main() {
50    let args: Vec<String> = env::args().collect();
51    if args.len() != 2 {
52        usage_error("expected exactly one input file.");
53    }
54
55    let input = Path::new(&args[1]);
56    let is_lol = match input.extension() {
57        Some(ext) => ext.to_string_lossy().eq_ignore_ascii_case("lol"),
58        None => false,
59    };
60    if !is_lol {
61        usage_error(&format!("'{}' is not a .lol file.", args[1]));
62    }
63
64    let source = match fs::read_to_string(input) {
65        Ok(s) => s,
66        Err(e) => {
67            eprintln!("Error: could not read '{}': {}", args[1], e);
68            process::exit(1);
69        }
70    };
71
72    // compile() exits on any lexical, syntax or semantic error
73    let mut compiler = LolCompiler::new("");
74    compiler.compile(&source);
75
76    let output = input.with_extension("html");
77    if let Err(e) = fs::write(&output, &compiler.html) {
78        eprintln!("Error: could not write '{}': {}", output.display(), e);
79        process::exit(1);
80    }
81    println!("Compiled successfully: {}", output.display());
82    open_in_browser(&output);
83}
